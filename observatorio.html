// --- GENERAR REPORTE --- function generarReporte() { Swal.fire({ title: 'Generando Reporte...', text: 'Por favor espere, esto puede tardar unos segundos.', allowOutsideClick: false, didOpen: () => Swal.showLoading() }); fetch(URL_APPS_SCRIPT, { method: 'POST', body: new URLSearchParams({ accion: 'generarReporte' }) }) .then(res => res.json()) .then(resp => { if (resp.exito) Swal.fire('¡Éxito!', resp.mensaje, 'success'); else Swal.fire('Atención', resp.mensaje, 'warning'); }) .catch(() => Swal.fire('Error', 'No se pudo conectar con el servidor.', 'error')); }
